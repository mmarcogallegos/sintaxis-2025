program CantidadVocales;

var
    cadena: string;
    cadenaAux: string;
    vocal:string;
    vocal2:string;
    vocal3:string;
    vocal4:string;
    vocal5:string;
    acumulador: real;
    posicion: real;
end;

begin

    vocal:="a";
    vocal2:="e";
    vocal3:="i";
    vocal4:="o";
    vocal5:="u";

    posicion := 1;
    acumulador := 0;

    read("ingrese cadena: ", cadena);
    
    while posicion <= LENGTH(cadena) do

        cadenaAux := EXTRACT(cadena, posicion, 1);
        
        if cadenaAux = vocal then
                acumulador := acumulador + 1;
        end;

        if cadenaAux = vocal2 then
                acumulador := acumulador + 1;
        end;
                
        if cadenaAux = vocal3 then
                acumulador := acumulador + 1;
        end;

        if cadenaAux = vocal4 then
                acumulador := acumulador + 1;
        end;

        if cadenaAux = vocal5 then
                acumulador := acumulador + 1;
        end;


        posicion := posicion + 1;

    end;

    write("Cantidad de vocales: ", acumulador);

end.
